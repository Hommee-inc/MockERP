# MockERP v0.1 — Núcleo Comercial (Tutorial completo)

```
Orçamento → Venda → Devolução
```

Antes de começar, confirme que você passou pelo `setup-ambiente-do-zero.md` e o ambiente virtual está ativado.

## O que você vai construir

Um script Python que gera, de forma coerente, clientes, produtos, orçamentos, vendas e devoluções, respeitando as regras de negócio (venda que nasce de orçamento, devolução que nunca ultrapassa o que foi vendido), e exporta tudo em Parquet.

## Passo 2: o código do gerador

Crie o arquivo `src/generators.py` com o conteúdo abaixo. Leia os comentários, eles explicam o porquê de cada parte, não só o que ela faz.

```python
import random
from datetime import timedelta
from faker import Faker

fake = Faker("pt_BR")
random.seed(42)  # fixa a aleatoriedade, para você conseguir reproduzir o mesmo dataset sempre que rodar


# ---------- Clientes e Produtos (base para tudo) ----------

def generate_customer(customer_id: int) -> dict:
    return {
        "customer_id": customer_id,
        "name": fake.name(),
        "document": fake.cpf(),
        "email": fake.email(),
        "city": fake.city(),
        "state": fake.estado_sigla(),
        "created_at": fake.date_time_between(start_date="-3y", end_date="now"),
        "status": random.choice(["ACTIVE", "INACTIVE"]),
    }


def generate_product(product_id: int) -> dict:
    return {
        "product_id": product_id,
        "sku": f"SKU-{product_id:06d}",
        "description": fake.catch_phrase(),
        "price": round(random.uniform(20, 500), 2),
    }


# ---------- Orçamento ----------

def generate_budget(budget_id: int, customer_ids: list[int]) -> dict:
    return {
        "budget_id": budget_id,
        "customer_id": random.choice(customer_ids),
        "budget_date": fake.date_time_between(start_date="-1y", end_date="now"),
        # weights: a maioria dos orçamentos converte em venda, o resto expira ou é rejeitado
        "status": random.choices(
            ["APPROVED_CONVERTED", "REJECTED", "EXPIRED"],
            weights=[60, 20, 20],
        )[0],
    }


def generate_budget_items(budget_id: int, products: list[dict]) -> list[dict]:
    n_items = random.randint(1, 5)
    chosen = random.sample(products, min(n_items, len(products)))
    items = []
    for i, product in enumerate(chosen):
        quantity = random.randint(1, 10)
        unit_price = product["price"]
        discount = round(unit_price * quantity * random.uniform(0, 0.1), 2)
        items.append({
            "budget_item_id": f"{budget_id}-{i+1:02d}",
            "budget_id": budget_id,
            "product_id": product["product_id"],
            "quantity": quantity,
            "unit_price": unit_price,
            "discount": discount,
            "total_amount": round(quantity * unit_price - discount, 2),
        })
    return items


# ---------- Venda (só existe para orçamentos convertidos) ----------

def generate_sale_from_budget(sale_id: int, budget: dict, budget_items: list[dict]) -> tuple[dict, list[dict]]:
    sale_date = budget["budget_date"] + timedelta(days=random.randint(1, 10))

    sale = {
        "sale_id": sale_id,
        "budget_id": budget["budget_id"],
        "customer_id": budget["customer_id"],
        "sale_date": sale_date,
        "status": "CONFIRMED",
        "total_amount": round(sum(i["total_amount"] for i in budget_items), 2),
    }

    # os itens da venda espelham os itens do orçamento aprovado, com IDs novos
    sale_items = []
    for i, bi in enumerate(budget_items):
        sale_items.append({
            "sale_item_id": f"{sale_id}-{i+1:02d}",
            "sale_id": sale_id,
            "product_id": bi["product_id"],
            "quantity": bi["quantity"],
            "unit_price": bi["unit_price"],
            "discount": bi["discount"],
            "total_amount": bi["total_amount"],
        })

    return sale, sale_items


# ---------- Devolução (respeita a correlação com a venda) ----------

def generate_return(return_id: int, sale: dict, sale_items: list[dict], return_rate: float = 0.15):
    """
    Retorna (None, []) quando essa venda não gera devolução.
    Quando gera, garante:
      - quantidade devolvida <= quantidade vendida por item
      - valor devolvido proporcional ao valor do item vendido
      - data da devolução sempre depois da data da venda
    """
    if random.random() > return_rate:
        return None, []

    return_date = sale["sale_date"] + timedelta(days=random.randint(1, 30))
    returned_items = []

    for i, si in enumerate(sale_items):
        if random.random() < 0.5:  # nem todo item da venda é devolvido
            continue

        max_qty = si["quantity"]
        returned_qty = random.randint(1, max_qty)  # nunca passa do que foi vendido
        proportion = returned_qty / si["quantity"]

        returned_items.append({
            "return_item_id": f"{return_id}-{i+1:02d}",
            "return_id": return_id,
            "sale_item_id": si["sale_item_id"],
            "quantity": returned_qty,
            "total_amount": round(proportion * si["total_amount"], 2),
        })

    if not returned_items:
        return None, []

    ret = {
        "return_id": return_id,
        "sale_id": sale["sale_id"],
        "return_date": return_date,
        "reason": random.choice(["DEFEITO", "ARREPENDIMENTO", "TROCA", "ENVIO_ERRADO"]),
        "status": "COMPLETED",
        "total_amount": round(sum(ri["total_amount"] for ri in returned_items), 2),
    }
    return ret, returned_items


# ---------- Orquestração: gera tudo junto, respeitando a ordem de dependência ----------

def generate_dataset(n_customers=1000, n_products=200, n_budgets=3000):
    customers = [generate_customer(i) for i in range(1, n_customers + 1)]
    customer_ids = [c["customer_id"] for c in customers]

    products = [generate_product(i) for i in range(1, n_products + 1)]

    budgets, budget_items_all = [], []
    sales, sale_items_all = [], []
    returns, return_items_all = [], []

    sale_id_counter = 1
    return_id_counter = 1

    for budget_id in range(1, n_budgets + 1):
        budget = generate_budget(budget_id, customer_ids)
        items = generate_budget_items(budget_id, products)
        budgets.append(budget)
        budget_items_all.extend(items)

        # só orçamento convertido vira venda, isso é a dependência entre as duas entidades
        if budget["status"] == "APPROVED_CONVERTED":
            sale, sale_items = generate_sale_from_budget(sale_id_counter, budget, items)
            sales.append(sale)
            sale_items_all.extend(sale_items)

            ret, return_items = generate_return(return_id_counter, sale, sale_items)
            if ret:
                returns.append(ret)
                return_items_all.extend(return_items)
                return_id_counter += 1

            sale_id_counter += 1

    return {
        "customers": customers,
        "products": products,
        "budgets": budgets,
        "budget_items": budget_items_all,
        "sales": sales,
        "sale_items": sale_items_all,
        "returns": returns,
        "return_items": return_items_all,
    }
```

**Por que a orquestração está em um loop só, e não em funções separadas por entidade rodando em sequência?** Porque a dependência real é por registro: uma venda específica só existe se aquele orçamento específico converteu. Gerar tudo em lotes separados (primeiro todos os orçamentos, depois todas as vendas) obrigaria a percorrer os orçamentos de novo para decidir quais viram venda. Gerar dentro do mesmo loop deixa a dependência explícita no código, o que ajuda bastante a entender o fluxo enquanto você está aprendendo.

## Passo 3: exportar para Parquet

Crie `src/export.py`:

```python
import pandas as pd
from pathlib import Path
from generators import generate_dataset

def export_dataset(dataset: dict, output_dir: str = "../data"):
    Path(output_dir).mkdir(parents=True, exist_ok=True)
    for name, records in dataset.items():
        df = pd.DataFrame(records)
        df.to_parquet(f"{output_dir}/{name}.parquet", engine="pyarrow", index=False)
        print(f"{name}: {len(df)} registros exportados")

if __name__ == "__main__":
    dataset = generate_dataset()
    export_dataset(dataset)
```

Rode (de dentro da pasta `src`):

```
cd src
python export.py
```

Você deve ver algo como:

```
customers: 1000 registros exportados
products: 200 registros exportados
budgets: 3000 registros exportados
budget_items: ~9000 registros exportados
sales: ~1800 registros exportados
sale_items: ~5400 registros exportados
returns: ~150 registros exportados
return_items: ~200 registros exportados
```

Os números exatos variam pouco por causa da aleatoriedade nas quantidades de item, mas com a seed fixa (`random.seed(42)`) o resultado deve ser sempre igual entre execuções.

## Passo 4: os testes

Crie `tests/test_generators.py`. Isso valida as regras de correlação que definimos (venda referencia cliente e orçamento válidos, devolução nunca ultrapassa o vendido):

```python
import sys
sys.path.append("../src")
from generators import generate_dataset


def test_every_sale_has_valid_customer():
    dataset = generate_dataset(n_customers=50, n_products=20, n_budgets=200)
    customer_ids = {c["customer_id"] for c in dataset["customers"]}
    for sale in dataset["sales"]:
        assert sale["customer_id"] in customer_ids


def test_every_sale_comes_from_converted_budget():
    dataset = generate_dataset(n_customers=50, n_products=20, n_budgets=200)
    converted_budget_ids = {
        b["budget_id"] for b in dataset["budgets"] if b["status"] == "APPROVED_CONVERTED"
    }
    for sale in dataset["sales"]:
        assert sale["budget_id"] in converted_budget_ids


def test_return_quantity_never_exceeds_sold_quantity():
    dataset = generate_dataset(n_customers=50, n_products=20, n_budgets=200)
    sale_item_qty = {si["sale_item_id"]: si["quantity"] for si in dataset["sale_items"]}
    for ri in dataset["return_items"]:
        assert ri["quantity"] <= sale_item_qty[ri["sale_item_id"]]


def test_return_amount_is_proportional():
    dataset = generate_dataset(n_customers=50, n_products=20, n_budgets=200)
    sale_items_by_id = {si["sale_item_id"]: si for si in dataset["sale_items"]}
    for ri in dataset["return_items"]:
        si = sale_items_by_id[ri["sale_item_id"]]
        # o valor devolvido nunca deve ultrapassar o valor total do item vendido
        assert ri["total_amount"] <= si["total_amount"] + 0.01  # tolerância de arredondamento
```

Rode os testes (de dentro da pasta `mockerp`, ambiente virtual ativado):

```
pytest tests/ -v
```

Você deve ver os quatro testes passando (`PASSED`). Se algum falhar, a mensagem do pytest mostra exatamente qual asserção quebrou, isso é o ponto de partida para debugar.
